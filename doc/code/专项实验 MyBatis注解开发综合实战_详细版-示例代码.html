1. 数据库脚本

sql/permission_schema.sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS mybatis_permission CHARACTER SET utf8mb4;
USE mybatis_permission;

-- 用户表
DROP TABLE IF EXISTS sys_user;
CREATE TABLE sys_user (
    user_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '用户ID',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名',
    password VARCHAR(100) NOT NULL COMMENT '密码',
    real_name VARCHAR(50) COMMENT '真实姓名',
    email VARCHAR(100) COMMENT '邮箱',
    phone VARCHAR(20) COMMENT '手机号',
    status INT DEFAULT 1 COMMENT '状态：0-禁用，1-正常',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    INDEX idx_username(username),
    INDEX idx_status(status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 角色表
DROP TABLE IF EXISTS sys_role;
CREATE TABLE sys_role (
    role_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '角色ID',
    role_name VARCHAR(50) NOT NULL UNIQUE COMMENT '角色名称',
    role_desc VARCHAR(200) COMMENT '角色描述',
    status INT DEFAULT 1 COMMENT '状态：0-禁用，1-正常',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    INDEX idx_role_name(role_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='角色表';

-- 权限表（支持树形结构）
DROP TABLE IF EXISTS sys_permission;
CREATE TABLE sys_permission (
    permission_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '权限ID',
    parent_id INT DEFAULT 0 COMMENT '父权限ID（0表示根节点）',
    permission_name VARCHAR(50) NOT NULL COMMENT '权限名称',
    permission_code VARCHAR(100) NOT NULL COMMENT '权限代码',
    permission_type INT COMMENT '权限类型：1-菜单，2-按钮，3-接口',
    url VARCHAR(200) COMMENT 'URL地址',
    icon VARCHAR(50) COMMENT '图标',
    sort INT DEFAULT 0 COMMENT '排序',
    status INT DEFAULT 1 COMMENT '状态：0-禁用，1-正常',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    INDEX idx_parent_id(parent_id),
    INDEX idx_code(permission_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='权限表';

-- 用户-角色关联表（多对多）
DROP TABLE IF EXISTS sys_user_role;
CREATE TABLE sys_user_role (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL COMMENT '用户ID',
    role_id INT NOT NULL COMMENT '角色ID',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    UNIQUE KEY uk_user_role (user_id, role_id),
    FOREIGN KEY (user_id) REFERENCES sys_user(user_id) ON DELETE CASCADE,
    FOREIGN KEY (role_id) REFERENCES sys_role(role_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户角色关联表';

-- 角色-权限关联表（多对多）
DROP TABLE IF EXISTS sys_role_permission;
CREATE TABLE sys_role_permission (
    id INT PRIMARY KEY AUTO_INCREMENT,
    role_id INT NOT NULL COMMENT '角色ID',
    permission_id INT NOT NULL COMMENT '权限ID',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    UNIQUE KEY uk_role_permission (role_id, permission_id),
    FOREIGN KEY (role_id) REFERENCES sys_role(role_id) ON DELETE CASCADE,
    FOREIGN KEY (permission_id) REFERENCES sys_permission(permission_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='角色权限关联表';

3. Mapper接口 - SysUserMapper

src/main/java/com/example/mapper/SysUserMapper.java
package com.example.mapper;

import com.example.pojo.SysUser;
import org.apache.ibatis.annotations.*;
import java.util.List;
import java.util.Map;

/**
 * 用户Mapper接口
 */
public interface SysUserMapper {
    
    // ==================== 基本CRUD ====================
    
    /**
     * 查询所有用户
     */
    @Select("SELECT * FROM sys_user WHERE status = 1")
    @Results(id = "userResultMap", value = {
        @Result(property = "userId", column = "user_id", id = true),
        @Result(property = "username", column = "username"),
        @Result(property = "password", column = "password"),
        @Result(property = "realName", column = "real_name"),
        @Result(property = "email", column = "email"),
        @Result(property = "phone", column = "phone"),
        @Result(property = "status", column = "status"),
        @Result(property = "createTime", column = "create_time"),
        @Result(property = "updateTime", column = "update_time")
    })
    List<SysUser> findAll();
    
    /**
     * 根据ID查询用户
     */
    @Select("SELECT * FROM sys_user WHERE user_id = #{userId} AND status = 1")
    @ResultMap("userResultMap")
    SysUser findById(Integer userId);
    
    /**
     * 根据用户名查询用户（用于登录）
     */
    @Select("SELECT * FROM sys_user WHERE username = #{username} AND status = 1")
    @ResultMap("userResultMap")
    SysUser findByUsername(String username);
    
    /**
     * 插入用户
     */
    @Insert("INSERT INTO sys_user(username, password, real_name, email, phone, status) " +
            "VALUES(#{username}, #{password}, #{realName}, #{email}, #{phone}, #{status})")
    @Options(useGeneratedKeys = true, keyProperty = "userId")
    int insert(SysUser user);
    
    /**
     * 更新用户
     */
    @Update({
        "<script>",
        "UPDATE sys_user",
        "<set>",
        "<if test='realName != null'>real_name = #{realName},</if>",
        "<if test='email != null'>email = #{email},</if>",
        "<if test='phone != null'>phone = #{phone},</if>",
        "<if test='status != null'>status = #{status},</if>",
        "</set>",
        "WHERE user_id = #{userId}",
        "</script>"
    })
    int update(SysUser user);
    
    /**
     * 更新密码
     */
    @Update("UPDATE sys_user SET password = #{password} WHERE user_id = #{userId}")
    int updatePassword(@Param("userId") Integer userId, 
                       @Param("password") String password);
    
    /**
     * 删除用户（逻辑删除）
     */
    @Update("UPDATE sys_user SET status = 0 WHERE user_id = #{userId}")
    int delete(Integer userId);
    
    // ==================== 多条件查询 ====================
    
    /**
     * 多条件查询用户（使用Provider）
     */
    @SelectProvider(type = com.example.provider.SysUserSqlProvider.class, method = "findByCondition")
    @ResultMap("userResultMap")
    List<SysUser> findByCondition(Map<String, Object> params);
    
    /**
     * 分页查询用户
     */
    @Select("SELECT * FROM sys_user WHERE status = 1 " +
            "ORDER BY create_time DESC LIMIT #{offset}, #{limit}")
    @ResultMap("userResultMap")
    List<SysUser> findByPage(@Param("offset") Integer offset, 
                               @Param("limit") Integer limit);
    
    /**
     * 查询用户总数
     */
    @Select("SELECT COUNT(*) FROM sys_user WHERE status = 1")
    int countTotal();
    
    // ==================== 关联查询 ====================
    
    /**
     * 查询用户及其角色
     */
    @Select("SELECT * FROM sys_user WHERE user_id = #{userId} AND status = 1")
    @Results({
        @Result(property = "userId", column = "user_id", id = true),
        @Result(property = "username", column = "username"),
        @Result(property = "roles", column = "user_id",
                many = @Many(select = "com.example.mapper.SysRoleMapper.findByUserId"))
    })
    SysUser findUserWithRoles(Integer userId);
    
    /**
     * 查询用户及其权限（通过角色获取）
     */
    @Select("SELECT * FROM sys_user WHERE user_id = #{userId} AND status = 1")
    @Results({
        @Result(property = "userId", column = "user_id", id = true),
        @Result(property = "permissions", column = "user_id",
                many = @Many(select = "com.example.mapper.SysPermissionMapper.findByUserId"))
    })
    SysUser findUserWithPermissions(Integer userId);
}

src/main/java/com/example/mapper/SysPermissionMapper.java
package com.example.mapper;

import com.example.pojo.SysPermission;
import org.apache.ibatis.annotations.*;
import java.util.List;

/**
 * 权限Mapper接口
 */
public interface SysPermissionMapper {
    
    /**
     * 查询所有权限
     */
    @Select("SELECT * FROM sys_permission WHERE status = 1 ORDER BY sort")
    @Results(id = "permissionResultMap", value = {
        @Result(property = "permissionId", column = "permission_id", id = true),
        @Result(property = "parentId", column = "parent_id"),
        @Result(property = "permissionName", column = "permission_name"),
        @Result(property = "permissionCode", column = "permission_code"),
        @Result(property = "permissionType", column = "permission_type"),
        @Result(property = "url", column = "url"),
        @Result(property = "icon", column = "icon"),
        @Result(property = "sort", column = "sort"),
        @Result(property = "status", column = "status"),
        @Result(property = "createTime", column = "create_time")
    })
    List<SysPermission> findAll();
    
    /**
     * 根据ID查询权限
     */
    @Select("SELECT * FROM sys_permission WHERE permission_id = #{permissionId} AND status = 1")
    @ResultMap("permissionResultMap")
    SysPermission findById(Integer permissionId);
    
    /**
     * 根据父ID查询子权限
     */
    @Select("SELECT * FROM sys_permission WHERE parent_id = #{parentId} AND status = 1 ORDER BY sort")
    @ResultMap("permissionResultMap")
    List<SysPermission> findByParentId(Integer parentId);
    
    /**
     * 根据用户ID查询权限（通过角色关联）
     */
    @Select("SELECT DISTINCT p.* FROM sys_permission p " +
            "INNER JOIN sys_role_permission rp ON p.permission_id = rp.permission_id " +
            "INNER JOIN sys_user_role ur ON rp.role_id = ur.role_id " +
            "WHERE ur.user_id = #{userId} AND p.status = 1 " +
            "ORDER BY p.sort")
    @ResultMap("permissionResultMap")
    List<SysPermission> findByUserId(Integer userId);
    
    /**
     * 根据角色ID查询权限
     */
    @Select("SELECT p.* FROM sys_permission p " +
            "INNER JOIN sys_role_permission rp ON p.permission_id = rp.permission_id " +
            "WHERE rp.role_id = #{roleId} AND p.status = 1 " +
            "ORDER BY p.sort")
    @ResultMap("permissionResultMap")
    List<SysPermission> findByRoleId(Integer roleId);
    
    /**
     * 查询权限树（递归查询子权限）
     */
    @Select("SELECT * FROM sys_permission WHERE parent_id = 0 AND status = 1 ORDER BY sort")
    @Results({
        @Result(property = "permissionId", column = "permission_id", id = true),
        @Result(property = "parentId", column = "parent_id"),
        @Result(property = "permissionName", column = "permission_name"),
        @Result(property = "children", column = "permission_id",
                many = @Many(select = "com.example.mapper.SysPermissionMapper.findChildren"))
    })
    List<SysPermission> findPermissionTree();
    
    /**
     * 递归查询子权限（内部调用）
     */
    @Select("SELECT * FROM sys_permission WHERE parent_id = #{permissionId} AND status = 1 ORDER BY sort")
    @Results({
        @Result(property = "permissionId", column = "permission_id", id = true),
        @Result(property = "parentId", column = "parent_id"),
        @Result(property = "permissionName", column = "permission_name"),
        @Result(property = "children", column = "permission_id",
                many = @Many(select = "com.example.mapper.SysPermissionMapper.findChildren"))
    })
    List<SysPermission> findChildren(Integer permissionId);
    
    /**
     * 插入权限
     */
    @Insert("INSERT INTO sys_permission(parent_id, permission_name, permission_code, " +
            "permission_type, url, icon, sort, status) " +
            "VALUES(#{parentId}, #{permissionName}, #{permissionCode}, " +
            "#{permissionType}, #{url}, #{icon}, #{sort}, #{status})")
    @Options(useGeneratedKeys = true, keyProperty = "permissionId")
    int insert(SysPermission permission);
    
    /**
     * 更新权限
     */
    @Update({
        "<script>",
        "UPDATE sys_permission",
        "<set>",
        "<if test='parentId != null'>parent_id = #{parentId},</if>",
        "<if test='permissionName != null'>permission_name = #{permissionName},</if>",
        "<if test='permissionCode != null'>permission_code = #{permissionCode},</if>",
        "<if test='permissionType != null'>permission_type = #{permissionType},</if>",
        "<if test='url != null'>url = #{url},</if>",
        "<if test='icon != null'>icon = #{icon},</if>",
        "<if test='sort != null'>sort = #{sort},</if>",
        "<if test='status != null'>status = #{status},</if>",
        "</set>",
        "WHERE permission_id = #{permissionId}",
        "</script>"
    })
    int update(SysPermission permission);
    
    /**
     * 删除权限（逻辑删除）
     */
    @Update("UPDATE sys_permission SET status = 0 WHERE permission_id = #{permissionId}")
    int delete(Integer permissionId);
}

/**
 * 用户SQL Provider类
 */
public class SysUserSqlProvider {
    
    /**
     * 动态查询用户
     */
    public String findByCondition(Map<String, Object> params) {
        StringBuilder sql = new StringBuilder();
        sql.append("SELECT * FROM sys_user WHERE status = 1 ");
        
        if (params.get("username") != null) {
            sql.append("AND username LIKE CONCAT('%', #{username}, '%') ");
        }
        if (params.get("realName") != null) {
            sql.append("AND real_name LIKE CONCAT('%', #{realName}, '%') ");
        }
        if (params.get("email") != null) {
            sql.append("AND email = #{email} ");
        }
        if (params.get("phone") != null) {
            sql.append("AND phone LIKE CONCAT('%', #{phone}, '%') ");
        }
        if (params.get("startTime") != null) {
            sql.append("AND create_time >= #{startTime} ");
        }
        if (params.get("endTime") != null) {
            sql.append("AND create_time <= #{endTime} ");
        }
        
        sql.append("ORDER BY create_time DESC");
        return sql.toString();
    }
}

package com.example.service;

import com.example.mapper.SysUserMapper;
import com.example.pojo.SysUser;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import java.io.InputStream;

/**
 * 登录服务类
 */
public class LoginService {
    
    /**
     * 用户登录验证
     * @param username 用户名
     * @param password 密码
     * @return 登录成功返回用户对象，失败返回null
     */
    public SysUser login(String username, String password) {
        try {
            // 1. 读取配置文件
            String resource = "mybatis-config.xml";
            InputStream inputStream = Resources.getResourceAsStream(resource);
            
            // 2. 创建SqlSessionFactory
            SqlSessionFactory sqlSessionFactory = new SqlSessionFactoryBuilder().build(inputStream);
            
            // 3. 创建SqlSession
            try (SqlSession session = sqlSessionFactory.openSession()) {
                // 4. 获取Mapper接口
                SysUserMapper mapper = session.getMapper(SysUserMapper.class);
                
                // 5. 根据用户名查询用户
                SysUser user = mapper.findByUsername(username);
                
                // 6. 验证密码
                if (user != null && password.equals(user.getPassword())) {
                    return user;
                }
                
                return null;
            }
        } catch (Exception e) {
            e.printStackTrace();
            return null;
        }
    }
}

package com.example.service;

import com.example.mapper.SysUserMapper;
import com.example.pojo.SysUser;
import java.util.List;
import java.util.stream.Collectors;

/**
 * 权限服务类
 */
public class PermissionService {
    
    private SysUserMapper userMapper;
    
    /**
     * 检查用户是否拥有某个权限
     * @param userId 用户ID
     * @param permissionCode 权限代码
     * @return 是否拥有权限
     */
    public boolean hasPermission(Integer userId, String permissionCode) {
        SysUser user = userMapper.findUserWithPermissions(userId);
        
        if (user == null || user.getPermissions() == null) {
            return false;
        }
        
        return user.getPermissions().stream()
                .anyMatch(p -> permissionCode.equals(p.getPermissionCode()));
    }
    
    /**
     * 获取用户的所有权限代码
     * @param userId 用户ID
     * @return 权限代码列表
     */
    public List<String> getUserPermissionCodes(Integer userId) {
        SysUser user = userMapper.findUserWithPermissions(userId);
        
        if (user == null || user.getPermissions() == null) {
            return List.of();
        }
        
        return user.getPermissions().stream()
                .map(p -> p.getPermissionCode())
                .collect(Collectors.toList());
    }
}
