sql/association_schema.sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS mybatis_association CHARACTER SET utf8mb4;
USE mybatis_association;

-- 创建用户表
DROP TABLE IF EXISTS user;
CREATE TABLE user (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '用户ID',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名',
    password VARCHAR(100) NOT NULL COMMENT '密码',
    age INT DEFAULT 0 COMMENT '年龄',
    email VARCHAR(100) COMMENT '邮箱',
    phone VARCHAR(20) COMMENT '手机号',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    deleted INT DEFAULT 0 COMMENT '删除标记：0-未删除，1-已删除'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 创建身份证表（一对一关系）
DROP TABLE IF EXISTS id_card;
CREATE TABLE id_card (
    card_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '身份证ID',
    user_id INT UNIQUE NOT NULL COMMENT '用户ID',
    card_no VARCHAR(18) NOT NULL COMMENT '身份证号',
    address VARCHAR(200) COMMENT '地址',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    INDEX idx_user_id(user_id),
    FOREIGN KEY (user_id) REFERENCES user(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='身份证表';

-- 创建订单表（一对多关系）
DROP TABLE IF EXISTS orders;
CREATE TABLE orders (
    order_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '订单ID',
    user_id INT NOT NULL COMMENT '用户ID',
    total_price DECIMAL(10,2) NOT NULL COMMENT '总金额',
    status INT DEFAULT 0 COMMENT '状态：0-待支付，1-已支付，2-已发货，3-已完成，4-已取消',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    INDEX idx_user_id(user_id),
    INDEX idx_status(status),
    FOREIGN KEY (user_id) REFERENCES user(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单表';

-- 创建学生表（多对多关系）
DROP TABLE IF EXISTS student;
CREATE TABLE student (
    student_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '学生ID',
    student_name VARCHAR(50) NOT NULL COMMENT '学生姓名',
    age INT COMMENT '年龄'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='学生表';

-- 创建课程表
DROP TABLE IF EXISTS course;
CREATE TABLE course (
    course_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '课程ID',
    course_name VARCHAR(100) NOT NULL COMMENT '课程名称',
    teacher VARCHAR(50) COMMENT '教师'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='课程表';

-- 创建选课表（中间表）
DROP TABLE IF EXISTS student_course;
CREATE TABLE student_course (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT 'ID',
    student_id INT NOT NULL COMMENT '学生ID',
    course_id INT NOT NULL COMMENT '课程ID',
    score DECIMAL(5,2) COMMENT '成绩',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    UNIQUE KEY uk_student_course (student_id, course_id),
    FOREIGN KEY (student_id) REFERENCES student(student_id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES course(course_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='选课表';

package com.example.mapper;

import com.example.pojo.User;
import org.apache.ibatis.annotations.*;
import java.util.List;

/**
 * 用户Mapper接口
 */
public interface UserMapper {
    
    /**
     * 根据ID查询用户
     */
    @Select("SELECT * FROM user WHERE id = #{id} AND deleted = 0")
    @Results(id = "userResultMap", value = {
        @Result(property = "id", column = "id", id = true),
        @Result(property = "username", column = "username"),
        @Result(property = "password", column = "password"),
        @Result(property = "age", column = "age"),
        @Result(property = "email", column = "email"),
        @Result(property = "phone", column = "phone"),
        @Result(property = "createTime", column = "create_time"),
        @Result(property = "updateTime", column = "update_time"),
        @Result(property = "deleted", column = "deleted")
    })
    User findById(Integer id);
    
    /**
     * 查询用户及其身份证（一对一关联）
     */
    @Select("SELECT * FROM user WHERE id = #{id} AND deleted = 0")
    @Results({
        @Result(property = "id", column = "id", id = true),
        @Result(property = "username", column = "username"),
        @Result(property = "idCard", column = "id",
                one = @One(select = "com.example.mapper.IdCardMapper.findByUserId",
                           fetchType = FetchType.LAZY))
    })
    User findUserWithIdCard(Integer id);
}

package com.example.mapper;

import com.example.pojo.IdCard;

 */
public interface IdCardMapper {

     * 根据ID查询身份证
     */

    /**
     * 查询身份证及其对应的用户（一对一关联）

package com.example.mapper;

import com.example.pojo.Orders;
import org.apache.ibatis.annotations.*;
import java.util.List;

/**
 * 订单Mapper接口
 */
public interface OrdersMapper {
    
    /**
     * 根据ID查询订单
     */
    @Select("SELECT * FROM orders WHERE order_id = #{orderId}")
    @Results(id = "orderResultMap", value = {
        @Result(property = "orderId", column = "order_id", id = true),
        @Result(property = "userId", column = "user_id"),
        @Result(property = "totalPrice", column = "total_price"),
        @Result(property = "status", column = "status"),
        @Result(property = "createTime", column = "create_time"),
        @Result(property = "updateTime", column = "update_time")
    })
    Orders findById(Integer orderId);
    
    /**
     * 根据用户ID查询订单
     */
    @Select("SELECT * FROM orders WHERE user_id = #{userId} ORDER BY create_time DESC")
    @ResultMap("orderResultMap")
    List<Orders> findByUserId(Integer userId);
    
    /**
     * 查询订单及其用户（多对一关联）
     */
    @Select("SELECT * FROM orders WHERE order_id = #{orderId}")
    @Results({
        @Result(property = "orderId", column = "order_id", id = true),
        @Result(property = "userId", column = "user_id"),
        @Result(property = "totalPrice", column = "total_price"),
        @Result(property = "user", column = "user_id",
                one = @One(select = "com.example.mapper.UserMapper.findById"))
    })
    Orders findOrderWithUser(Integer orderId);
}

package com.example.mapper;

import com.example.pojo.Student;
import org.apache.ibatis.annotations.*;
import java.util.List;

/**
 * 学生Mapper接口
 */
public interface StudentMapper {
    
    /**
     * 查询所有学生
     */
    @Select("SELECT * FROM student")
    @Results(id = "studentResultMap", value = {
        @Result(property = "studentId", column = "student_id", id = true),
        @Result(property = "studentName", column = "student_name"),
        @Result(property = "age", column = "age")
    })
    List<Student> findAll();
    
    /**
     * 根据ID查询学生
     */
    @Select("SELECT * FROM student WHERE student_id = #{studentId}")
    @ResultMap("studentResultMap")
    Student findById(Integer studentId);
    
    /**
     * 查询学生及其所选课程（多对多关联）
     */
    @Select("SELECT * FROM student WHERE student_id = #{studentId}")
    @Results({
        @Result(property = "studentId", column = "student_id", id = true),
        @Result(property = "studentName", column = "student_name"),
        @Result(property = "courses", column = "student_id",
                many = @Many(select = "com.example.mapper.CourseMapper.findByStudentId"))
    })
    Student findStudentWithCourses(Integer studentId);
    
    /**
     * 根据课程ID查询选课学生
     */
    @Select("SELECT s.* FROM student s " +
            "INNER JOIN student_course sc ON s.student_id = sc.student_id " +
            "WHERE sc.course_id = #{courseId}")
    @ResultMap("studentResultMap")
    List<Student> findByCourseId(Integer courseId);
}

        System.out.println("课程：" + course.getCourseName());
        System.out.println("教师：" + course.getTeacher());
        
        List<Student> students = course.getStudents();
        System.out.println("选课学生数：" + students.size());
        
        for (Student student : students) {
            System.out.println("  学生：" + student.getStudentName() + 
                             "，年龄：" + student.getAge());
        }
        
        assert course != null : "课程不应该为null";
    }
}

8. 注意事项

  1. 确保数据库 mybatis_association 已创建并导入测试数据
  2. 修改 mybatis-config.xml 中的数据库连接信息
  3. 一对一关联查询时，避免双向关联都设置为立即加载（会导致无限递归）
  4. 对于不常用的关联数据，建议使用延迟加载（FetchType.LAZY）
  5. 观察SQL日志，理解嵌套查询（N+1问题）和延迟加载的效果
