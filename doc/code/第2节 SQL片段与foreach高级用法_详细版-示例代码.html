案例示例代码 - 第2节 SQL片段与foreach高级用法

本节内容概要：本节提供MyBatis SQL片段（sql/include）与foreach循环两大高级特性的完整示例代码。涵盖可复用SQL片段定义、foreach批量操作（INSERT/UPDATE/IN查询）、嵌套foreach、多表批量关联等企业级开发常用场景。

一、sql标签与include标签 - SQL片段复用

sql标签核心原理：<sql id="片段ID">用于定义可复用的SQL片段，<include refid="片段ID">用于引用该片段。通过此机制，可以将公共字段、公共条件抽取出来，实现DRY（Don't Repeat Yourself）原则，减少维护成本。

1.1 基础用法 - 公共字段抽取

UserMapper.xml - SQL片段基础用法

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!-- ==================== SQL片段定义区 ==================== -->

    <!-- 示例1-1：基础SQL片段 - 用户表公共字段 -->
    <sql id="userColumns">
        id, username, password, email, age, gender, address, create_time, status
    </sql>

    <!-- 示例1-2：带别名的SQL片段 - 用于关联查询 -->
    <sql id="userColumnsWithAlias">
        u.id, u.username, u.email, u.age, u.gender, u.address, u.status
    </sql>

    <!-- 示例1-3：带条件的SQL片段 - 公共WHERE条件 -->
    <sql id="activeUserWhere">
        WHERE status = 1
    </sql>

    <!-- 示例1-4：复杂SQL片段 - 排序语句 -->
    <sql id="commonOrderBy">
        ORDER BY id DESC, create_time DESC
    </sql>

    <!-- ==================== 查询语句使用片段 ==================== -->

    <!-- 示例1-5：使用SQL片段查询所有用户 -->
    <select id="findAll" resultType="com.example.mybatis.entity.User">
        SELECT <include refid="userColumns"/>
        FROM tb_user
        <include refid="activeUserWhere"/>
    </select>

    <!-- 示例1-6：使用SQL片段 + where条件 -->
    <select id="findActiveUsers" resultType="com.example.mybatis.entity.User">
        SELECT <include refid="userColumns"/>
        FROM tb_user
        <include refid="activeUserWhere"/>
        AND age >= #{minAge}
        <include refid="commonOrderBy"/>
    </select>

    <!-- 示例1-7：关联查询使用带别名的SQL片段 -->
    <select id="findAllWithAlias" resultType="com.example.mybatis.entity.User">
        SELECT <include refid="userColumnsWithAlias"/>
        FROM tb_user u
        <include refid="activeUserWhere"/>
    </select>

    <!-- 示例1-8：UPDATE语句中使用SQL片段 -->
    <update id="updateBaseInfo">
        UPDATE tb_user
        SET
            email = #{email},
            address = #{address},
            update_time = NOW()
        WHERE id = #{id}
    </update>

    <!-- 示例1-9：INSERT语句中复用字段片段 -->
    <insert id="insertUser" useGeneratedKeys="true" keyProperty="id">
        INSERT INTO tb_user(
            <include refid="userColumns"/>
        ) VALUES (
            #{id}, #{username}, #{password}, #{email}, #{age}, #{gender},
            #{address}, #{createTime}, #{status}
        )
    </insert>

</mapper>

1.2 SQL片段进阶 - 参数化片段与动态片段

SQL片段的include与refid属性详解

<!--
    include标签的refid属性支持以下写法：
    1. refid="片段ID" - 引用当前mapper中的片段
    2. refid="命名空间.片段ID" - 引用其他mapper中的片段
    3. ${属性名} - 使用${}在refid中动态拼接（谨慎使用，防SQL注入）
    4. #{属性名} - 不支持，refid必须是字符串字面量
-->

<!-- 示例1-10：引用其他Mapper的SQL片段（跨Mapper复用） -->
<!-- 在UserMapper.xml中： -->
<select id="findUsersWithDept" resultType="UserVO">
    SELECT <include refid="userColumnsWithAlias"/>,
           <include refid="com.example.mybatis.mapper.DeptMapper.deptColumns"/>
    FROM tb_user u
    LEFT JOIN tb_dept d ON u.dept_id = d.id
    WHERE u.status = 1
</select>

<!-- 示例1-11：SQL片段中嵌套include（片段引用片段） -->
<sql id="fullUserColumns">
    <include refid="userColumnsWithAlias"/>,
    <include refid="com.example.mybatis.mapper.DeptMapper.deptColumns"/>
</sql>

<!-- 示例1-12：动态SQL片段（片段中包含if标签） -->
<sql id="dynamicWhereClause">
    <where>
        <if test="status != null">
            AND status = #{status}
        </if>
        <if test="minAge != null">
            AND age >= #{minAge}
        </if>
        <if test="keyword != null and keyword != ''">
            AND (username LIKE CONCAT('%', #{keyword}, '%')
                 OR email LIKE CONCAT('%', #{keyword}, '%'))
        </if>
    </where>
</sql>

<!-- 引用动态片段 -->
<select id="dynamicSearch" resultType="User">
    SELECT <include refid="userColumns"/>
    FROM tb_user
    <include refid="dynamicWhereClause"/>
    <include refid="commonOrderBy"/>
</select>

<!-- 示例1-13：SQL片段中使用${}动态字段名（需严格校验，防注入） -->
<sql id="dynamicOrderBy">
    ORDER BY ${sortField} ${sortOrder}
</sql>

<select id="orderByField" resultType="User">
    SELECT <include refid="userColumns"/>
    FROM tb_user
    <include refid="activeUserWhere"/>
    <if test="sortField != null and sortField != ''">
        <include refid="dynamicOrderBy"/>
    </if>
</select>

二、foreach标签详解 - 循环迭代与批量操作

foreach标签核心属性：
collection：必填，要迭代的集合/数组/Map（List、Set、Map、数组）
item：必填，循环迭代时的当前元素别名，在标签体内通过#{item}引用
index：可选，当前迭代的索引（从0开始），Map迭代时为key
open：可选，迭代开始前拼接的字符（如"("）
separator：可选，每次迭代之间的分隔符（如","、" OR "）
close：可选，迭代结束后拼接的字符（如")"）

2.1 foreach基础 - IN查询

UserMapper.java - foreach基础接口

package com.example.mybatis.mapper;

import com.example.mybatis.entity.User;
import org.apache.ibatis.annotations.Param;
import java.util.List;
import java.util.Set;
import java.util.Map;

public interface UserMapper {

    /**
     * 批量查询用户 - List参数
     */
    List<User> findByIds(@Param("ids") List<Long> ids);

    /**
     * 批量查询用户 - Set参数
     */
    List<User> findByIdSet(@Param("ids") Set<Long> ids);

    /**
     * 批量查询用户 - 数组参数
     */
    List<User> findByIdArray(@Param("ids") Long[] ids);

    /**
     * 批量查询多状态用户
     */
    List<User> findByStatuses(@Param("statuses") List<Integer> statuses);

    /**
     * Map参数遍历 - key为索引
     */
    List<User> findByConditionsMap(@Param("params") Map<String, Object> params);
}

UserMapper.xml - foreach IN查询示例

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!-- ==================== foreach基础示例 ==================== -->

    <!-- 示例2-1：foreach基础 - List的IN查询 (1,2,3,4,5) -->
    <select id="findByIds" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, gender, address, status
        FROM tb_user
        WHERE status = 1
        AND id IN
        <!--
            open: 开头拼接 "("
            close: 结尾拼接 ")"
            separator: 每个元素之间用 "," 分隔
        -->
        <foreach collection="ids" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
        <!-- 最终生成: WHERE id IN (1, 2, 3, 4, 5) -->
    </select>

    <!-- 示例2-2：foreach - Set的IN查询 -->
    <select id="findByIdSet" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, status
        FROM tb_user
        WHERE id IN
        <foreach collection="ids" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
    </select>

    <!-- 示例2-3：foreach - 数组的IN查询 -->
    <select id="findByIdArray" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, status
        FROM tb_user
        WHERE id IN
        <foreach collection="ids" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
    </select>

    <!-- 示例2-4：foreach - 多状态IN查询（status IN (1, 2)） -->
    <select id="findByStatuses" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, status
        FROM tb_user
        WHERE status IN
        <foreach collection="statuses" item="status" open="(" separator="," close=")">
            #{status}
        </foreach>
        ORDER BY id DESC
    </select>

    <!-- 示例2-5：foreach - Map遍历（key为字段名，value为值） -->
    <select id="findByConditionsMap" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, status
        FROM tb_user
        WHERE 1 = 1
        <!-- params是一个Map: {username: "张三", status: 1} -->
        <foreach collection="params" index="key" item="value">
            AND ${key} = #{value}
        </foreach>
        <!-- 等价于: WHERE username = '张三' AND status = 1 -->
    </select>

    <!-- 示例2-6：foreach - 字符串集合IN查询 -->
    <select id="findByUsernames" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, status
        FROM tb_user
        WHERE username IN
        <foreach collection="usernames" item="name" open="(" separator="," close=")">
            #{name}
        </foreach>
        <!-- 如果用${}会导致: WHERE username IN ('zhangsan', 'lisi')
             如果用#{}会导致: WHERE username IN (?, ?)
             字符串类型推荐用#{name} -->
    </select>

</mapper>

2.2 foreach进阶 - 批量INSERT

UserMapper.java - 批量操作接口

package com.example.mybatis.mapper;

import com.example.mybatis.entity.User;
import org.apache.ibatis.annotations.Param;
import java.util.List;

public interface UserMapper {

    /**
     * 批量插入用户 - 返回插入条数
     * INSERT INTO tb_user (cols) VALUES (vals1), (vals2), (vals3)
     */
    int batchInsertUsers(@Param("users") List<User> users);

    /**
     * 批量插入用户 - 返回主键列表（MySQL）
     * useGeneratedKeys + keyProperty 自动回填主键
     */
    int batchInsertUsersWithKeys(@Param("users") List<User> users);

    /**
     * 批量更新用户
     */
    int batchUpdateUsers(@Param("users") List<User> users);

    /**
     * 批量删除用户
     */
    int batchDeleteUsers(@Param("ids") List<Long> ids);
}

UserMapper.xml - 批量INSERT详解

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!--
        foreach批量INSERT注意事项：
        1. MySQL默认允许最大包大小为4MB，可通过max_allowed_packet调整
        2. 建议单次批量不超过1000条，超出时分批处理
        3. 使用VALUES多行插入语法，效率远高于逐条INSERT
        4. useGeneratedKeys + keyProperty 只返回第一条的主键，需手动处理
    -->

    <!-- 示例2-7：foreach批量INSERT - 基础版 -->
    <insert id="batchInsertUsers">
        INSERT INTO tb_user(
            username, password, email, age, gender, address, status, create_time
        ) VALUES
        <foreach collection="users" item="user" separator=",">
            (
                #{user.username},
                #{user.password},
                #{user.email},
                #{user.age},
                #{user.gender},
                #{user.address},
                #{user.status},
                NOW()
            )
        </foreach>
        <!--
            生成SQL示例：
            INSERT INTO tb_user(cols) VALUES
            ('zhangsan', 'pass1', 'z1@e.com', 20, 1, '北京', 1, NOW()),
            ('lisi', 'pass2', 'l2@e.com', 22, 0, '上海', 1, NOW())
        -->
    </insert>

    <!-- 示例2-8：foreach批量INSERT - 自动回填主键版（MySQL） -->
    <insert id="batchInsertUsersWithKeys" useGeneratedKeys="true"
            keyProperty="id" databaseId="mysql">
        INSERT INTO tb_user(
            username, password, email, age, gender, address, status, create_time
        ) VALUES
        <foreach collection="users" item="user" separator=",">
            (
                #{user.username},
                #{user.password},
                #{user.email},
                #{user.age},
                #{user.gender},
                #{user.address},
                #{user.status},
                NOW()
            )
        </foreach>
        <!-- 重要：useGeneratedKeys只能回填第一个对象的主键
             如果需要回填所有对象的主键，需要使用Oracle序列方式
             或在Java代码中手动处理 -->
    </insert>

    <!-- 示例2-9：foreach批量INSERT - Oracle序列版本 -->
    <insert id="batchInsertUsersOracle" databaseId="oracle">
        INSERT INTO tb_user(
            id, username, password, email, age, gender, address, status
        )
        <foreach collection="users" item="user" separator="SELECT 1 FROM DUAL UNION ALL SELECT ">
            SELECT SEQ_USER.NEXTVAL, #{user.username}, #{user.password},
                   #{user.email}, #{user.age}, #{user.gender},
                   #{user.address}, #{user.status}
            FROM DUAL
        </foreach>
    </insert>

    <!-- 示例2-10：foreach批量INSERT - 包含selectKey获取序列 -->
    <insert id="batchInsertWithSeq" databaseId="oracle">
        <selectKey keyProperty="id" resultType="long" order="BEFORE">
            SELECT NVL(MAX(id), 0) + 1 FROM tb_user
        </selectKey>
        INSERT INTO tb_user(
            id, username, password, email, age, gender, address, status
        ) VALUES
        <foreach collection="users" item="user" separator=",">
            (
                #{user.id},
                #{user.username},
                #{user.password},
                #{user.email},
                #{user.age},
                #{user.gender},
                #{user.address},
                #{user.status}
            )
        </foreach>
    </insert>

</mapper>

2.3 foreach进阶 - 批量UPDATE与DELETE

UserMapper.xml - 批量UPDATE与DELETE示例

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!-- 示例2-11：foreach批量UPDATE - CASE WHEN方式（推荐，一条SQL） -->
    <update id="batchUpdateUsers">
        UPDATE tb_user
        SET update_time = NOW()
        <foreach collection="users" item="user">
            , #{user.fieldName} = CASE id
                <foreach collection="users" item="u">
                    WHEN #{u.id} THEN #{u.fieldValue}
                </foreach>
            END
        </foreach>
        WHERE id IN
        <foreach collection="users" item="user" open="(" separator="," close=")">
            #{user.id}
        </foreach>
    </update>

    <!-- 示例2-12：foreach批量UPDATE - 简洁版（每条记录只更新相同字段） -->
    <update id="batchUpdateStatus">
        UPDATE tb_user
        SET status = #{newStatus},
            update_time = NOW()
        WHERE id IN
        <foreach collection="ids" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
    </update>

    <!-- 示例2-13：foreach批量UPDATE - 批量修改密码 -->
    <update id="batchUpdatePassword">
        UPDATE tb_user
        SET password =
        <foreach collection="users" item="user" separator="" close="">
            (CASE id
                <foreach collection="users" item="u">
                    WHEN #{u.id} THEN #{u.newPassword}
                </foreach>
            END)
        </foreach>
        WHERE id IN
        <foreach collection="ids" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
        <!-- 注意：此写法会导致UPDATE语句中只有最后一个user生效
             实际生产中建议按批次逐条更新或使用存储过程 -->
    </update>

    <!-- 示例2-14：foreach批量DELETE - 方式一：IN删除 -->
    <delete id="batchDeleteUsers">
        DELETE FROM tb_user
        WHERE id IN
        <foreach collection="ids" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
        <!-- 生成: DELETE FROM tb_user WHERE id IN (1, 2, 3) -->
    </delete>

    <!-- 示例2-15：foreach批量DELETE - 方式二：多行DELETE -->
    <delete id="batchDeleteUsersMultiRow">
        <foreach collection="ids" item="id" separator=";" close=";">
            DELETE FROM tb_user WHERE id = #{id}
        </foreach>
        <!-- 生成: DELETE FROM tb_user WHERE id = 1; DELETE FROM tb_user WHERE id = 2; ...
             注意：MyBatis默认只支持单语句执行，需开启allowMultiQueries=true -->
    </delete>

    <!-- 示例2-16：foreach批量DELETE - 方式三：动态条件删除 -->
    <delete id="batchDeleteByConditions">
        DELETE FROM tb_user
        WHERE
        <foreach collection="users" item="user" separator=" OR ">
            (id = #{user.id} AND username = #{user.username})
        </foreach>
        <!-- 生成: DELETE FROM tb_user WHERE
             (id = 1 AND username = 'zhangsan')
             OR (id = 2 AND username = 'lisi') -->
    </delete>

</mapper>

三、foreach高级用法 - 复杂场景实战

3.1 嵌套foreach - Map遍历

嵌套foreach详解 - Map<String, List<Object>>

<!--
    Map遍历场景：
    1. Map<String, Object> - index为key，item为value
    2. Map<String, List<Object>> - 外层遍历key，内层遍历List
    3. List<Map<String, Object>> - 外层遍历List，内层遍历Map
-->

<!-- 示例3-1：Map<String, Object> 遍历 - 动态SET -->
<update id="updateFields">
    UPDATE tb_user
    SET update_time = NOW()
    <foreach collection="params" index="field" item="value" separator="">
        , ${field} = #{value}
    </foreach>
    WHERE id = #{id}
    <!--
        调用示例：
        Map<String, Object> params = new HashMap<>();
        params.put("email", "new@example.com");
        params.put("address", "北京市");
        mapper.updateFields(1L, params);
        -->
</update>

<!-- 示例3-2：List<Map<String, Object>> 遍历 - 批量INSERT动态字段 -->
<insert id="batchInsertDynamic">
    INSERT INTO tb_user_tag(user_id, tag_name, tag_value, create_time)
    VALUES
    <foreach collection="tagList" item="tag" separator=",">
        (
            #{tag.userId},
            #{tag.tagName},
            #{tag.tagValue},
            NOW()
        )
    </foreach>
    <!-- tagList: List<Map<String, Object>>
         每个Map的key: userId, tagName, tagValue -->
</insert>

<!-- 示例3-3：嵌套foreach - Map<String, List<String>> -->
<select id="findByMultiCategories" resultType="Product">
    SELECT id, product_name, category, price, stock
    FROM tb_product
    WHERE 1 = 1
    <!-- outerMap: Map<String, List<String>>
         外层遍历: index="category", item="values"
         内层遍历: collection="values", item="val"
    -->
    <foreach collection="outerMap" index="category" item="values">
        AND (
            <foreach collection="values" item="val" separator=" OR ">
                category = #{category} AND product_name LIKE CONCAT('%', #{val}, '%')
            </foreach>
        )
    </foreach>
    <!-- outerMap示例: {"电子产品": ["手机", "电脑"], "食品": ["零食"]}
         生成: WHERE 1=1
         AND ((category='电子产品' AND name LIKE '%手机%') OR (category='电子产品' AND name LIKE '%电脑%'))
         AND ((category='食品' AND name LIKE '%零食%')) -->
</select>

<!-- 示例3-4：嵌套foreach - 多List的笛卡尔积查询 -->
<select id="findCombination" resultType="Product">
    SELECT DISTINCT p.id, p.product_name, p.category, p.price
    FROM tb_product p
    WHERE p.status = 1
    AND (
        <foreach collection="categories" item="cat" separator=" OR ">
            p.category = #{cat}
        </foreach>
    )
    AND (
        <foreach collection="keywords" item="kw" separator=" OR ">
            p.product_name LIKE CONCAT('%', #{kw}, '%')
        </foreach>
    )
    <!-- categories: ["电子产品", "食品"]
         keywords: ["手机", "零食"]
         相当于: WHERE category IN ('电子产品','食品')
                 AND (name LIKE '%手机%' OR name LIKE '%零食%') -->
</select>

3.2 foreach与动态SQL组合 - 复杂查询场景

UserMapper.xml - 复杂动态查询

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!-- 示例3-5：动态IN + 多条件组合查询 -->
    <select id="complexSearch" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, gender, address, status
        FROM tb_user
        WHERE 1 = 1
        <!-- IN查询：按用户名列表查询 -->
        <if test="usernames != null and usernames.size() > 0">
            AND username IN
            <foreach collection="usernames" item="name" open="(" separator="," close=")">
                #{name}
            </foreach>
        </if>
        <!-- IN查询：按状态列表查询 -->
        <if test="statuses != null and statuses.size() > 0">
            AND status IN
            <foreach collection="statuses" item="s" open="(" separator="," close=")">
                #{s}
            </foreach>
        </if>
        <!-- IN查询：按年龄段列表查询（age IN (18, 25, 30)） -->
        <if test="ages != null and ages.size() > 0">
            AND age IN
            <foreach collection="ages" item="age" open="(" separator="," close=")">
                #{age}
            </foreach>
        </if>
        <!-- 关键词模糊匹配 -->
        <if test="keyword != null and keyword != ''">
            AND (username LIKE CONCAT('%', #{keyword}, '%')
                 OR email LIKE CONCAT('%', #{keyword}, '%'))
        </if>
    </select>

    <!-- 示例3-6：foreach + choose 动态模式切换 -->
    <select id="searchWithMode" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, status
        FROM tb_user
        WHERE 1 = 1
        <choose>
            <!-- 模式A：批量ID查询 -->
            <when test="ids != null and ids.size() > 0">
                AND id IN
                <foreach collection="ids" item="id" open="(" separator="," close=")">
                    #{id}
                </foreach>
            </when>
            <!-- 模式B：批量用户名查询 -->
            <when test="usernames != null and usernames.size() > 0">
                AND username IN
                <foreach collection="usernames" item="name" open="(" separator="," close=")">
                    #{name}
                </foreach>
            </when>
            <!-- 模式C：默认条件 -->
            <otherwise>
                AND status = 1
            </otherwise>
        </choose>
        ORDER BY id DESC
    </select>

    <!-- 示例3-7：foreach + where + set 综合实战 -->
    <select id="queryForUpdate" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, status
        FROM tb_user
        <where>
            <if test="status != null and status >= 0">
                AND status = #{status}
            </if>
            <if test="minAge != null">
                AND age >= #{minAge}
            </if>
            <if test="ids != null and ids.size() > 0">
                AND id IN
                <foreach collection="ids" item="id" open="(" separator="," close=")">
                    #{id}

            </if>

3.3 批量操作性能优化

批量操作最佳实践

package com.example.mybatis.service;

import com.example.mybatis.entity.User;
import com.example.mybatis.mapper.UserMapper;
import com.example.mybatis.util.SqlSessionUtil;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;

import java.util.ArrayList;
import java.util.List;

/**
 * 批量操作最佳实践
 */
public class BatchOperationService {

    private SqlSessionFactory sqlSessionFactory;

    public BatchOperationService(SqlSessionFactory factory) {
        this.sqlSessionFactory = factory;
    }

    /**
     * 最佳实践1：大批量分批处理（防止SQL过长/内存溢出）
     * 每次处理500条，MySQL的max_allowed_packet默认4MB
     */
    public int batchInsertLargeData(List<User> users) {
        int total = 0;
        int batchSize = 500;
        List<User> batch = new ArrayList<>();

        for (int i = 0; i < users.size(); i++) {
            batch.add(users.get(i));

            // 每batchSize条执行一次
            if (batch.size() == batchSize || i == users.size() - 1) {
                try (SqlSession session = sqlSessionFactory.openSession(false)) {
                    UserMapper mapper = session.getMapper(UserMapper.class);
                    int rows = mapper.batchInsertUsers(batch);
                    session.commit();
                    total += rows;
                    System.out.println("批次[" + (i / batchSize + 1) + "] 插入 " + rows + " 条");
                    batch.clear();
                } catch (Exception e) {
                    System.err.println("批次插入失败: " + e.getMessage());
                    throw new RuntimeException(e);
                }
            }
        }
        return total;
    }

    /**
     * 最佳实践2：批量更新（按批次，每批500条）
     */
    public int batchUpdateLargeData(List<User> users) {
        int total = 0;
        int batchSize = 500;
        List<User> batch = new ArrayList<>();

        for (int i = 0; i < users.size(); i++) {
            batch.add(users.get(i));

            if (batch.size() == batchSize || i == users.size() - 1) {
                try (SqlSession session = sqlSessionFactory.openSession(false)) {
                    UserMapper mapper = session.getMapper(UserMapper.class);
                    int rows = mapper.batchUpdateUsers(batch);
                    session.commit();
                    total += rows;
                    batch.clear();
                }
            }
        }
        return total;
    }

    /**
     * 最佳实践3：使用ExecutorType.BATCH批处理模式（最高效）
     * 将多次INSERT合并为一次网络往返
     */
    public int batchInsertWithBatchExecutor(List<User> users) {
        // BATCH模式：所有SQL先缓存，最后flush时一起发送
        try (SqlSession session = sqlSessionFactory.openSession(ExecutorType.BATCH)) {
            UserMapper mapper = session.getMapper(UserMapper.class);

            int total = 0;
            for (User user : users) {
                mapper.batchInsertUsers(List.of(user)); // 每次只放1条
                total++;
                // 每500条flush一次，防止内存溢出
                if (total % 500 == 0) {
                    session.flushStatements();
                }
            }
            session.commit();
            return total;
        }
    }

    /**
     * 最佳实践4：批量删除（分批处理）
     */
    public int batchDeleteLargeData(List<Long> ids) {
        int total = 0;
        int batchSize = 500;
        List<Long> batch = new ArrayList<>();

        for (int i = 0; i < ids.size(); i++) {
            batch.add(ids.get(i));

            if (batch.size() == batchSize || i == ids.size() - 1) {
                try (SqlSession session = sqlSessionFactory.openSession(false)) {
                    UserMapper mapper = session.getMapper(UserMapper.class);
                    int rows = mapper.batchDeleteUsers(batch);
                    session.commit();
                    total += rows;
                    batch.clear();
                }
            }
        }
        return total;
    }
}
