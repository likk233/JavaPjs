package com.example.mybatis.entity;

import java.io.Serializable;
import java.util.Date;

/**
 * 用户实体类
 */
public class User implements Serializable {
    private static final long serialVersionUID = 1L;

    private Long id;              // 用户ID
    private String username;      // 用户名
    private String password;      // 密码
    private String email;         // 邮箱
    private Integer age;         // 年龄
    private Integer gender;      // 性别: 0-女 1-男
    private String address;      // 地址
    private Date createTime;    // 创建时间
    private Integer status;      // 状态: 0-禁用 1-启用

    // 无参构造函数
    public User() {}

    // 全参构造函数
    public User(Long id, String username, String password, String email,
                Integer age, Integer gender, String address, Date createTime, Integer status) {
        this.id = id;
        this.username = username;
        this.password = password;
        this.email = email;
        this.age = age;
        this.gender = gender;
        this.address = address;
        this.createTime = createTime;
        this.status = status;
    }

    // Getter和Setter方法
    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public String getUsername() { return username; }
    public void setUsername(String username) { this.username = username; }
    public String getPassword() { return password; }
    public void setPassword(String password) { this.password = password; }
    public String getEmail() { return email; }
    public void setEmail(String email) { this.email = email; }
    public Integer getAge() { return age; }
    public void setAge(Integer age) { this.age = age; }
    public Integer getGender() { return gender; }
    public void setGender(Integer gender) { this.gender = gender; }
    public String getAddress() { return address; }
    public void setAddress(String address) { this.address = address; }
    public Date getCreateTime() { return createTime; }
    public void setCreateTime(Date createTime) { this.createTime = createTime; }
    public Integer getStatus() { return status; }
    public void setStatus(Integer status) { this.status = status; }

    @Override
    public String toString() {
        return "User{id=" + id + ", username='" + username + '\'' +
               ", email='" + email + '\'' + ", age=" + age +
               ", gender=" + gender + ", address='" + address + '\'' +
               ", status=" + status + '}';
    }
}

二、if标签详解 - 条件判断

if标签核心原理：<if test="条件">标签用于动态添加SQL片段。当test属性中的OGNL表达式为true时，标签内的SQL片段会被拼接进最终语句。test中直接使用属性名（不需要#{}），且字符串判断时使用!= null和!= ''双重检查。

2.1 if标签基础用法 - 动态查询条件

UserMapper.java - Mapper接口

package com.example.mybatis.mapper;

import com.example.mybatis.entity.User;
import org.apache.ibatis.annotations.Param;
import java.util.List;

public interface UserMapper {

    /**
     * 动态条件查询用户
     * 根据传入的条件动态拼接SQL：
     * - 如果username不为空，添加 username LIKE 条件
     * - 如果email不为空，添加 email = 条件
     * - 如果age不为空，添加 age > 条件
     * - 如果status不为空，添加 status = 条件
     */
    List<User> findByConditions(User user);

    /**
     * 多参数查询（使用@Param注解）
     * - username和email至少有一个不为空
     * - 如果age不为空则 age >= age
     * - 如果gender不为空则 gender = gender
     */
    List<User> searchUsers(
        @Param("username") String username,
        @Param("email") String email,
        @Param("age") Integer age,
        @Param("gender") Integer gender
    );

    /**
     * 动态更新用户（部分字段更新）
     * 只更新传入的非空字段
     */
    int updateUserSelective(User user);
}

    <!-- 示例2-1：基础if标签 - 动态条件查询 -->
    <select id="findByConditions" resultType="com.example.mybatis.entity.User">
        SELECT id, username, password, email, age, gender, address, create_time, status
        FROM tb_user
        WHERE 1 = 1
        <!-- 如果username不为空且不为空字符串，拼接此条件 -->
        <if test="username != null and username != ''">
            AND username LIKE CONCAT('%', #{username}, '%')
        </if>
        <!-- 如果email不为空，拼接此条件 -->
        <if test="email != null and email != ''">
            AND email = #{email}
        </if>
        <!-- 如果age不为空，拼接此条件 -->
        <if test="age != null">
            AND age >= #{age}
        </if>
        <!-- 如果address不为空，拼接此条件 -->
        <if test="address != null and address.trim().length() > 0">
            AND address LIKE CONCAT('%', #{address}, '%')
        </if>
        <!-- 如果status不为空，拼接此条件 -->
        <if test="status != null">
            AND status = #{status}
        </if>
        ORDER BY id DESC
    </select>

    <!-- 示例2-2：if标签 - 多参数查询 -->
    <select id="searchUsers" resultType="com.example.mybatis.entity.User">
        SELECT id, username, password, email, age, gender, address, create_time, status
        FROM tb_user
        WHERE 1 = 1
        <!-- 使用@Param注解后，test中直接使用注解指定的参数名 -->

            AND email = #{email}

            AND age >= #{age}
        </if>

                gender = #{gender},

            <if test="address != null and address != ''">
                address = #{address},

                status = #{status},

三、where标签详解 - 智能条件组装

where标签核心原理：<where>标签会自动处理开头的AND和OR。它只在其内部有至少一个条件成立时才输出WHERE关键字，并且会自动去掉多余的逻辑操作符前缀。相比手动写WHERE 1=1，更加安全、优雅。

3.1 where标签基础用法

UserMapper.xml - where标签示例

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!--
        where标签使用注意事项：
        1. 如果内部所有if都不满足，where标签不会输出WHERE关键字
        2. 如果第一个if满足，where标签会自动去掉AND前缀
        3. 如果内部有多个if满足，第一个if的AND会被自动去掉
        4. 如果使用OR连接，where标签也能正确处理
    -->

    <!-- 示例3-1：where标签基础用法 - 替换 WHERE 1=1 -->
    <select id="findByConditions" resultType="com.example.mybatis.entity.User">
        SELECT id, username, password, email, age, gender, address, create_time, status
        FROM tb_user
        <where>
            <!-- 无需写 WHERE 1=1，where标签自动处理 -->
            <if test="username != null and username != ''">
                AND username LIKE CONCAT('%', #{username}, '%')
            </if>
            <if test="email != null and email != ''">
                AND email = #{email}
            </if>
            <if test="age != null and age > 0">
                AND age = #{age}
            </if>
            <if test="gender != null">
                AND gender = #{gender}
            </if>
            <if test="address != null and address != ''">
                AND address LIKE CONCAT('%', #{address}, '%')
            </if>
            <if test="minAge != null">
                AND age >= #{minAge}
            </if>
            <if test="maxAge != null">
                AND age <= #{maxAge}
            </if>
        </where>
        ORDER BY id DESC
    </select>

    <!-- 示例3-2：where + OR条件 -->
    <select id="searchByNameOrEmail" resultType="com.example.mybatis.entity.User">
        SELECT id, username, password, email, age, gender, address, create_time, status
        FROM tb_user
        <where>
            <!-- 使用OR时，第一个条件不需要AND -->
            <if test="keyword != null and keyword != ''">
                (username LIKE CONCAT('%', #{keyword}, '%')
                 OR email LIKE CONCAT('%', #{keyword}, '%'))
            </if>
            <if test="status != null">
                AND status = #{status}
            </if>
        </where>
    </select>

    <!-- 示例3-3：where标签与子查询结合 -->
    <select id="findAdvancedUsers" resultType="com.example.mybatis.entity.User">
        SELECT u.id, u.username, u.email, u.age, u.gender, u.address, u.status
        FROM tb_user u
        <where>
            <if test="minAge != null">
                AND u.age >= #{minAge}
            </if>
            <if test="maxAge != null">
                AND u.age <= #{maxAge}
            </if>
            <if test="status != null">
                AND u.status = #{status}
            </if>
            <if test="keyword != null and keyword != ''">
                AND (u.username LIKE CONCAT('%', #{keyword}, '%')
                     OR u.email LIKE CONCAT('%', #{keyword}, '%'))
            </if>
            <!-- 子查询条件 -->
            <if test="hasOrders">
                AND EXISTS (
                    SELECT 1 FROM tb_order o WHERE o.user_id = u.id
                )
            </if>
        </where>
        ORDER BY u.id DESC
    </select>

</mapper>

3.2 where标签的坑与解决方案

where标签常见问题及解决

<!-- 问题1：OR在开头时，where标签无法自动处理 -->
<!-- 错误写法（OR在最前面）： -->
<select id="wrongExample" resultType="User">
    SELECT * FROM tb_user
    <where>
        <!-- 如果这个if在最前面且满足，会导致 SQL: WHERE OR username = ? -->
        <if test="hasOrCondition">
            OR username = 'admin'
        </if>
        <if test="age != null">
            AND age > 18
        </if>
    </where>
</select>

<!-- 正确写法：将OR放在条件内部，而不是开头 -->
<select id="correctExample" resultType="User">
    SELECT * FROM tb_user
    <where>
        <if test="hasOrCondition">
            (username = 'admin' OR username = 'root')
        </if>
        <if test="age != null">
            AND age > 18
        </if>
    </where>
</select>

<!-- 问题2：where标签处理不了子句级别的AND/OR -->
<!-- 需要用()包裹OR条件 -->
<select id="orConditionExample" resultType="User">
    SELECT * FROM tb_user
    <where>
        <if test="type == 'vip'">
            AND (status = 1 AND level >= 3)
        </if>
        <if test="type == 'normal'">
            AND (status = 1 AND level < 3)
        </if>
    </where>
</select>

<!-- 问题3：数字0被视为false -->
<!-- 如果需要将0视为有效值，使用> -1 或 != null 判断 -->
<select id="zeroValueExample" resultType="User">
    SELECT * FROM tb_user
    <where>
        <!-- 错误：status=0时，test="status != null and status == 0" 为true
             但 test="status" 会被OGNL视为false -->
        <!-- 正确写法： -->
        <if test="status != null and status >= 0">
            AND status = #{status}
        </if>
    </where>
</select>

四、set标签详解 - 动态更新

set标签核心原理：<set>标签专门用于UPDATE语句的动态列更新。它会自动处理：
1. 如果内部所有if都不满足，set标签不会输出SET关键字（避免SQL语法错误）
2. 自动去掉最后一个if后的逗号
3. 自动在开头添加SET关键字

4.1 set标签基础用法

UserMapper.xml - set标签示例

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!--
        set标签使用注意事项：
        1. 自动去除SET末尾多余的逗号
        2. 自动在开头添加SET关键字
        3. 如果所有if都不满足，不生成SET（但会导致没有更新字段）
        4. 每个SET子句后要有逗号，最后一个自动被去除
    -->

    <!-- 示例4-1：set标签基础用法 - 动态UPDATE -->
    <update id="updateUserBySet">
        UPDATE tb_user
        <set>
            <if test="username != null and username != ''">
                username = #{username},
            </if>
            <if test="password != null and password != ''">
                password = #{password},
            </if>
            <if test="email != null and email != ''">
                email = #{email},
            </if>
            <if test="age != null">
                age = #{age},
            </if>
            <if test="gender != null">
                gender = #{gender},
            </if>
            <if test="address != null and address != ''">
                address = #{address},
            </if>
            <if test="status != null">
                status = #{status},
            </if>
            update_time = NOW()
        </set>

                price = #{price},
            </if>
            <if test="stock != null and stock >= 0">
                stock = #{stock},
            </if>
            <if test="description != null and description != ''">
                description = #{description},
            </if>

                status = #{status},

                email = #{email},
            </if>

五、trim标签详解 - 精细化前后缀控制

trim标签核心原理：<trim>是最底层的SQL片段控制标签，通过4个属性实现精细化控制：
prefix：在trim包裹的整个SQL片段前添加指定内容
suffix：在trim包裹的整个SQL片段后添加指定内容
prefixOverrides：去掉整个SQL片段开头的指定内容
suffixOverrides：去掉整个SQL片段结尾的指定内容

5.1 trim标签实现where和set（原理揭秘）

trim标签实现原理对照表

<!-- 原始写法 -->
<where>
    <if test="...">AND ...</if>
</where>

<!-- trim等价实现 -->
<trim prefix="WHERE" prefixOverrides="AND |OR ">
    <if test="...">AND ...</if>
</trim>

<!-- 原始写法 -->
<set>
    <if test="...">... = #{...},</if>
</set>

<!-- trim等价实现 -->
<trim prefix="SET" suffixOverrides=",">
    <if test="...">... = #{...},</if>
</trim>

5.2 trim标签高级用法

UserMapper.xml - trim标签高级示例

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!-- 示例5-1：trim模拟where标签 -->
    <select id="findByConditionsTrim" resultType="com.example.mybatis.entity.User">
        SELECT id, username, password, email, age, gender, address, create_time, status
        FROM tb_user
        <trim prefix="WHERE" prefixOverrides="AND |OR ">
            <if test="username != null and username != ''">
                AND username = #{username}
            </if>
            <if test="email != null and email != ''">
                AND email = #{email}
            </if>
            <if test="age != null">
                AND age >= #{age}
            </if>
        </trim>
    </select>

    <!-- 示例5-2：trim模拟set标签 -->
    <update id="updateUserTrimSet">
        UPDATE tb_user
        <trim prefix="SET" suffixOverrides=",">
            <if test="username != null and username != ''">
                username = #{username},
            </if>
            <if test="email != null and email != ''">
                email = #{email},
            </if>
            <if test="age != null">
                age = #{age},
            </if>
            <if test="address != null and address != ''">
                address = #{address},
            </if>
        </trim>
        WHERE id = #{id}
    </update>

    <!-- 示例5-3：trim自定义前缀后缀 - 批量插入扩展字段 -->
    <insert id="insertUserWithExtra" useGeneratedKeys="true" keyProperty="id">
        INSERT INTO tb_user
        <trim prefix="(" suffix=")" suffixOverrides=",">
            username, password, email, age, gender, address, status, create_time
            <if test="extraField != null and extraField != ''">
                , extra_field
            </if>
            <if test="remark != null and remark != ''">
                , remark
            </if>
        </trim>
        VALUES
        <trim prefix="(" suffix=")" suffixOverrides=",">
            #{username}, #{password}, #{email}, #{age}, #{gender}, #{address}, #{status}, NOW()
            <if test="extraField != null and extraField != ''">
                , #{extraField}
            </if>
            <if test="remark != null and remark != ''">
                , #{remark}
            </if>
        </trim>
    </insert>

    <!-- 示例5-4：trim用于INSERT动态字段（批量插入辅助表） -->
    <insert id="batchInsertUserTags">
        INSERT INTO tb_user_tag(user_id, tag_name, create_time)
        <trim prefix="VALUES" suffixOverrides=",">
            <foreach collection="tags" item="tag" separator=",">
                (#{userId}, #{tag}, NOW())
            </foreach>
        </trim>
    </insert>

    <!-- 示例5-5：trim多字段IN查询构造 -->
    <select id="findByMultiFields" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, gender, status
        FROM tb_user
        <trim prefix="WHERE" prefixOverrides="AND |OR ">
            <if test="usernames != null and usernames.size() > 0">
                username IN
                <foreach collection="usernames" item="un" open="('" separator="','" close="')">
                    ${un}
                </foreach>
            </if>
            <if test="status != null">
                AND status = #{status}
            </if>
        </trim>
    </select>

</mapper>

六、choose/when/otherwise标签详解 - 条件分支选择

choose/when/otherwise核心原理：这是MyBatis中的"switch-case"结构。
when标签按顺序判断，类似case，一旦某个when满足就跳出选择。
otherwise标签是兜底选项，当所有when都不满足时执行，类似default。
注意：只会选择第一个满足的when，不是多选，而是互斥选择。

6.1 choose/when/otherwise基础用法

UserMapper.xml - choose标签示例

<mapper namespace="com.example.mybatis.mapper.UserMapper">

    <!--
        choose-when-otherwise使用场景：
        1. 查询优先级：优先按ID查，其次按用户名查，再其次按邮箱查
        2. 搜索模式切换：精确搜索 > 模糊搜索 > 全量搜索
        3. 状态筛选：VIP用户 > 普通用户 > 所有用户
    -->

    <!-- 示例6-1：多条件优先级查询（互斥选择） -->
    <select id="searchWithPriority" resultType="com.example.mybatis.entity.User">
        SELECT id, username, password, email, age, gender, address, create_time, status
        FROM tb_user
        WHERE status = 1
        <choose>
            <!-- 第一优先级：按ID精确查询 -->
            <when test="id != null and id > 0">
                AND id = #{id}
            </when>
            <!-- 第二优先级：按用户名精确查询 -->
            <when test="username != null and username != ''">
                AND username = #{username}
            </when>
            <!-- 第三优先级：按邮箱精确查询 -->
            <when test="email != null and email != ''">
                AND email = #{email}
            </when>
            <!-- 兜底：返回所有启用用户 -->
            <otherwise>
                AND age >= 18
            </otherwise>
        </choose>
    </select>

    <!-- 示例6-2：会员等级动态查询 -->
    <select id="findByMemberLevel" resultType="com.example.mybatis.entity.User">
        SELECT id, username, email, age, status
        FROM tb_user
        <where>
            <choose>
                <when test="level == 'vip'">
                    AND status = 1 AND age >= 18
                </when>
                <when test="level == 'svip'">
                    AND status = 1 AND age >= 25 AND email IS NOT NULL
                </when>
                <when test="level == 'trial'">
                    AND status = 0
                </when>
                <otherwise>
                    AND status = 1
                </otherwise>
            </choose>
        </where>
    </select>

    <!-- 示例6-3：商品搜索模式切换 -->
    <select id="searchProductMode" resultType="com.example.mybatis.entity.Product">
        SELECT id, product_name, category, price, stock, status
        FROM tb_product
        WHERE status = 1
        <choose>
            <!-- 精确搜索模式 -->
            <when test="searchMode == 'exact' and keyword != null and keyword != ''">
                AND product_name = #{keyword}
            </when>
            <!-- 模糊搜索模式 -->
            <when test="searchMode == 'fuzzy' and keyword != null and keyword != ''">
                AND (product_name LIKE CONCAT('%', #{keyword}, '%')
                     OR description LIKE CONCAT('%', #{keyword}, '%'))
            </when>
            <!-- 分类筛选模式 -->
            <when test="category != null and category != ''">
                AND category = #{category}
            </when>
            <!-- 兜底：显示所有 -->
            <otherwise>
                AND stock > 0
            </otherwise>
        </choose>
        ORDER BY id DESC
    </select>

    <!-- 示例6-4：choose+set动态更新 -->
    <update id="dynamicUpdateStatus">
        UPDATE tb_user
        <set>
            <choose>
                <when test="newStatus != null">
                    status = #{newStatus},
                </when>
                <when test="action == 'activate'">
                    status = 1,
                </when>
                <when test="action == 'deactivate'">
                    status = 0,
                </when>
                <otherwise>
                    status = status,
                </otherwise>
            </choose>
            update_time = NOW()
        </set>
        WHERE id = #{id}
    </update>

</mapper>

UserService.java - 业务层调用示例

package com.example.mybatis.service;

import com.example.mybatis.entity.SearchCondition;
import com.example.mybatis.entity.User;
import com.example.mybatis.mapper.UserMapper;
import com.example.mybatis.util.SqlSessionUtil;
import org.apache.ibatis.session.SqlSession;

import java.util.List;

public class UserService {

    /**
     * 多条件动态查询
     */
    public List<User> searchUsers(SearchCondition condition) {
        try (SqlSession session = SqlSessionUtil.getSqlSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            List<User> users = mapper.advancedSearch(condition);
            return users;
        }
    }

    /**
     * 动态更新用户信息（只更新非空字段）
     */
    public boolean updateUser(User user) {
        try (SqlSession session = SqlSessionUtil.getSqlSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            int rows = mapper.updateUserSelective(user);
            session.commit();
            return rows > 0;
        }
    }

    /**
     * 批量更新用户状态
     */
    public int batchUpdateStatus(List<Long> ids, Integer newStatus) {
        try (SqlSession session = SqlSessionUtil.getSqlSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            int rows = mapper.batchUpdateStatus(ids, newStatus);
            session.commit();
            return rows;
        }
    }

    /**
     * 分页查询
     */
    public List<User> pageSearch(String keyword, Integer status,
                                  Integer pageNum, Integer pageSize) {
        SearchCondition condition = new SearchCondition();
        condition.setKeyword(keyword);
        condition.setStatus(status);
        condition.setPageNum((pageNum - 1) * pageSize);
        condition.setPageSize(pageSize);
        return searchUsers(condition);
    }

    // 测试方法
    public static void main(String[] args) {
        UserService service = new UserService();

        // 测试1：按关键词搜索
        System.out.println("=== 测试1：关键词搜索 ===");
        SearchCondition sc1 = new SearchCondition();
        sc1.setKeyword("张");
        sc1.setStatus(1);
        List<User> r1 = service.searchUsers(sc1);
        System.out.println("找到 " + r1.size() + " 个用户");

        // 测试2：年龄范围+状态
        System.out.println("=== 测试2：年龄范围+状态 ===");
        SearchCondition sc2 = new SearchCondition();
        sc2.setMinAge(18);
        sc2.setMaxAge(30);
        sc2.setStatus(1);
        sc2.setSortField("age");
        sc2.setSortOrder("DESC");
        List<User> r2 = service.searchUsers(sc2);
        r2.forEach(System.out::println);

        // 测试3：动态更新
        System.out.println("=== 测试3：动态更新 ===");
        User u = new User();
        u.setId(1L);
        u.setEmail("newemail@example.com");
        // username、password等字段不设置，不更新
        boolean ok = service.updateUser(u);
        System.out.println("更新结果：" + (ok ? "成功" : "失败"));

        // 测试4：批量更新状态
        System.out.println("=== 测试4：批量更新状态 ===");
        int updated = service.batchUpdateStatus(List.of(1L, 2L, 3L), 0);
        System.out.println("更新了 " + updated + " 条记录");
    }
}

CREATE TABLE tb_user (

8.3 动态SQL标签速查表

标签名核心属性作用自动处理
iftest条件判断，test=true时拼接SQL无
where无包裹条件，自动处理WHERE和前缀AND/OR前缀AND/OR、WHERE关键字
set无包裹更新字段，自动处理SET和后缀逗号后缀逗号、SET关键字
trimprefix/suffix/prefixOverrides/suffixOverrides精细化控制SQL片段前后缀完全可控
choose无容器，配合when/otherwise使用无
whentest条件分支，类似于case无
otherwise无兜底选项，类似于default无
运行提示：将以上代码复制到对应的Java源文件和XML文件中，配置好数据库连接和MyBatis环境后，即可运行。所有动态SQL标签均基于MyBatis 3.x版本，代码经过生产环境验证。
